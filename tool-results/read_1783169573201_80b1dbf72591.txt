   158→function formatDuration(ms: number) {
   159→  const s = Math.round(ms / 1000);
   160→  if (s < 60) return `${s}s`;
   161→  const m = Math.floor(s / 60);
   162→  return `${m}m ${s % 60}s`;
   163→}
   164→
   165→function formatRelativeTime(iso: string) {
   166→  const then = new Date(iso).getTime();
   167→  const now = Date.now();
   168→  const diff = Math.max(0, now - then);
   169→  const sec = Math.floor(diff / 1000);
   170→  if (sec < 60) return `${sec}s ago`;
   171→  const min = Math.floor(sec / 60);
   172→  if (min < 60) return `${min}m ago`;
   173→  const hr = Math.floor(min / 60);
   174→  if (hr < 24) return `${hr}h ago`;
   175→  const day = Math.floor(hr / 24);
   176→  if (day < 7) return `${day}d ago`;
   177→  return new Date(iso).toLocaleDateString();
   178→}

   794→function CountriesTab() {
   795→  const [countries, setCountries] = useState<Country[]>([]);
   796→  const [loading, setLoading] = useState(true);
   797→  const [dialogOpen, setDialogOpen] = useState(false);

   799→  const [form, setForm] = useState<Country>(emptyCountry);
   800→  const [saving, setSaving] = useState(false);
   801→  const [confirmDelete, setConfirmDelete] = useState<Country | null>(null);
   802→
   803→  const fetchCountries = useCallback(async () => {
   804→    setLoading(true);
   805→    try {
   806→      const res = await fetch("/api/admin/countries", { cache: "no-store" });
   807→      if (!res.ok) throw new Error("Failed");
   808→      const json = (await res.json()) as { countries: Country[] };
   809→      setCountries(json.countries);
   810→    } catch {
   811→      toast.error("Could not load countries.");
   812→    } finally {
   813→      setLoading(false);
   814→    }
   815→  }, []);
   816→
   817→  useEffect(() => {
   818→    fetchCountries();
   819→  }, [fetchCountries]);
   820→
   821→  const openAdd = () => {
   822→    setEditing(null);
   823→    setForm({ ...emptyCountry, order: countries.length });
   824→    setDialogOpen(true);
   825→  };
   826→
   827→  const openEdit = (c: Country) => {
   828→    setEditing(c);
   829→    setForm({ ...c });
   830→    setDialogOpen(true);
   831→  };
   832→
   833→  const updateField = <K extends keyof Country>(key: K, value: Country[K]) =>
   834→    setForm((f) => ({ ...f, [key]: value }));
   835→
   836→  const submitCountry = async () => {
   837→    if (!form.slug || !form.name || !form.flagEmoji) {
   838→      toast.error("Slug, name and flag emoji are required.");
   839→      return;
   840→    }
   841→    setSaving(true);
   842→    try {
   843→      const isEdit = !!editing && !!editing.id;
   844→      const body = isEdit ? { id: editing!.id, ...form } : form;
   845→      const res = await fetch("/api/admin/countries", {
   846→        method: isEdit ? "PUT" : "POST",
   847→        headers: { "Content-Type": "application/json" },
   848→        body: JSON.stringify(body),
   849→      });
   850→      if (!res.ok) {
   851→        const err = await res.json().catch(() => ({}));
   852→        throw new Error(err.error ?? "Save failed");
   853→      }
   854→      toast.success(isEdit ? "Country updated." : "Country created.");
   855→      setDialogOpen(false);
   856→      await fetchCountries();
   857→    } catch (e) {
   858→      toast.error(e instanceof Error ? e.message : "Save failed.");
   859→    } finally {
   860→      setSaving(false);
   861→    }

   864→  const deleteCountry = async (c: Country) => {
   865→    if (!c.id) return;

   868→        method: "DELETE",
   869→      });
   870→      if (!res.ok) throw new Error("Delete failed");
   871→      toast.success(`${c.name} deleted.`);
   872→      setConfirmDelete(null);
   873→      await fetchCountries();
   874→    } catch {
   875→      toast.error("Could not delete country.");
   876→    }

   896→          </Button>
   897→        </div>

   900→      <div className="rounded-2xl border border-border bg-card shadow-sm">
   901→        {loading ? (

   904→              <Skeleton key={i} className="h-12 w-full" />
