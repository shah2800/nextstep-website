   157→   157→
   158→   158→function formatDuration(ms: number) {
   159→   159→  const s = Math.round(ms / 1000);
   160→   160→  if (s < 60) return `${s}s`;
   161→   161→  const m = Math.floor(s / 60);
   162→   162→  return `${m}m ${s % 60}s`;
   163→   163→}
   164→   164→
   165→   165→function formatRelativeTime(iso: string) {
   166→   166→  const then = new Date(iso).getTime();
   167→   167→  const now = Date.now();
   168→   168→  const diff = Math.max(0, now - then);
   169→   169→  const sec = Math.floor(diff / 1000);
   170→   170→  if (sec < 60) return `${sec}s ago`;
   171→   171→  const min = Math.floor(sec / 60);
   172→   172→  if (min < 60) return `${min}m ago`;
   173→   173→  const hr = Math.floor(min / 60);
   174→   174→  if (hr < 24) return `${hr}h ago`;
   175→   175→  const day = Math.floor(hr / 24);
   176→   176→  if (day < 7) return `${day}d ago`;
   177→   177→  return new Date(iso).toLocaleDateString();
   178→   178→}

   491→   491→
   492→   492→function AnalyticsTab() {
   493→   493→  const [data, setData] = useState<AnalyticsData | null>(null);
   494→   494→  const [loading, setLoading] = useState(true);
   495→   495→  const [refreshing, setRefreshing] = useState(false);
   496→   496→
   497→   497→  const fetchAnalytics = useCallback(async (silent = false) => {
   498→   498→    if (silent) setRefreshing(true);
   499→   499→    else setLoading(true);
   500→   500→    try {
   501→   501→      const res = await fetch("/api/admin/analytics", { cache: "no-store" });
   502→   502→      if (!res.ok) throw new Error("Failed to load analytics");
   503→   503→      const json = (await res.json()) as AnalyticsData;
   504→   504→      setData(json);
   505→   505→    } catch {
   506→   506→      toast.error("Could not load analytics. Please try again.");
   507→   507→    } finally {
   508→   508→      setLoading(false);
   509→   509→      setRefreshing(false);
   510→   510→    }
