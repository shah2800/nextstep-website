   248→function formatDuration(ms: number) {
   249→  const s = Math.round(ms / 1000);
   250→  if (s < 60) return `${s}s`;
   251→  const m = Math.floor(s / 60);
   252→  return `${m}m ${s % 60}s`;
   253→}
   254→
   255→function formatRelativeTime(iso: string) {
   256→  const then = new Date(iso).getTime();
   257→  const now = Date.now();
   258→  const diff = Math.max(0, now - then);
   259→  const sec = Math.floor(diff / 1000);
   260→  if (sec < 60) return `${sec}s ago`;
   261→  const min = Math.floor(sec / 60);
   262→  if (min < 60) return `${min}m ago`;
   263→  const hr = Math.floor(min / 60);
   264→  if (hr < 24) return `${hr}h ago`;
   265→  const day = Math.floor(hr / 24);
   266→  if (day < 7) return `${day}d ago`;
   267→  return new Date(iso).toLocaleDateString();
   268→}
   269→
   270→function formatLeadDate(iso: string) {
   271→  try {
   272→    const d = new Date(iso);
   273→    if (Number.isNaN(d.getTime())) return iso;
   274→    return d.toLocaleString("en-US", {
   275→      month: "short",
   276→      day: "numeric",
   277→      hour: "numeric",
   278→      minute: "2-digit",
   279→      hour12: true,
   280→    });
   281→  } catch {
   282→    return iso;
   283→  }
   284→}
